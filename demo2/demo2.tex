\begin{proposition}[The Implicit Function Theorem]
    \review{What is $F$?}
    \(F: U \subset \mathbb{R}^{2} \to \mathbb{R}\),
    If $F(a, b) = 0$ and $F(x, y)$ is continuously differentiable on some open disk with center $(a, b)$, then, if
    $$\frac{\partial F}{\partial y}(a, b) \neq 0$$
    there exists an $h > 0$ and a unique function $\phi(x)$ continuously differentiable \review{What can we say about $\phi$?} defined for $|x - a| < h$ such that $\phi(a) = b$ and $F(x, \phi(x)) = 0$ for $|x - a| < h$.
\end{proposition}

\begin{proposition}[{\href{https://web.math.princeton.edu/~js129/PDFs/teaching/analysis_1_fall_2014/Colloquium_Week_9_Continuity_Pasting_Lemma.pdf}{The Actual Pasting Lemma}}]
    Let $X$ and $Y$ be metric spaces. Let $(A, B) \in \text{Closed}(X)^2$ and assume further that $X = A \cup B$. 
    Assume that we have two functions $f : A \to Y$ and $g : B \to Y$.
    Assume that $f(x) = g(x) \quad \forall x \in A \cap B$.
    Then we can define a new function $h : X \to Y$ by:
    \[
    h(x) = \begin{cases}
        f(x) & \text{if } x \in A, \\
        g(x) & \text{if } x \in B
    \end{cases}
    \]
    (if $x \in A \cap B$ then we have no ambiguity by assumption).
    \textbf{Claim:} If both $f$ and $g$ are continuous, then $h$ is continuous.
\end{proposition}

\begin{definition}
    Let $P\in\mathbb{R}[x,y]$ be a squarefree polynomial, with $\deg_{y}(P) > 0$. We denote by $\operatorname{proj}(P)$ the set of points $x\in\mathbb{R}$ such that either $\operatorname{Disc}_{y}(P)(x)=0$ or $\operatorname{lc}_{y}(P)(x)=0$.
\end{definition}
