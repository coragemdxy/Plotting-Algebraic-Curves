\begin{proposition}[The Implicit Function Theorem]
    \review{What is $F$?}
    \(F: U \subset \mathbb{R}^{2} \to \mathbb{R}\),
    If $F(a, b) = 0$ and $F(x, y)$ is continuously differentiable on some open disk with center $(a, b)$, then, if
    $$\frac{\partial F}{\partial y}(a, b) \neq 0$$
    there exists an $h > 0$ and a unique function $\phi(x)$ continuously differentiable \review{What can we say about $\phi$?} defined for $|x - a| < h$ such that $\phi(a) = b$ and $F(x, \phi(x)) = 0$ for $|x - a| < h$.
\end{proposition}

\begin{proposition}[{\href{https://web.math.princeton.edu/~js129/PDFs/teaching/analysis_1_fall_2014/Colloquium_Week_9_Continuity_Pasting_Lemma.pdf}{The Actual Pasting Lemma}}]
    Let $X$ and $Y$ be metric spaces. Let $(A, B) \in \text{Closed}(X)^2$ and assume further that $X = A \cup B$. 
    Assume that we have two functions $f : A \to Y$ and $g : B \to Y$.
    Assume that $f(x) = g(x) \quad \forall x \in A \cap B$.
    Then we can define a new function $h : X \to Y$ by:
    \[
    h(x) = \begin{cases}
        f(x) & \text{if } x \in A, \\
        g(x) & \text{if } x \in B
    \end{cases}
    \]
    (if $x \in A \cap B$ then we have no ambiguity by assumption).
    \textbf{Claim:} If both $f$ and $g$ are continuous, then $h$ is continuous.
\end{proposition}

\begin{definition}
    Let $P\in\mathbb{R}[x,y]$ be a squarefree polynomial, with $\deg_{y}(P) > 0$. We denote by $\operatorname{proj}(P)$ the set of points $x\in\mathbb{R}$ such that either $\operatorname{Disc}_{y}(P)(x)=0$ or $\operatorname{lc}_{y}(P)(x)=0$.
\end{definition}

\begin{exercise}
    Let $P \in \mathbb{R}[x,y]$ be a squarefree polynomial, with $\deg_{y}(P) > 0$. Suppose that $\#\operatorname{proj}(P)<\infty$. Let $x_{1},x_{2},\dots,x_{s}$ be the elements of $\operatorname{proj}(P)$ in ascending order, and let $x_{0} = -\infty$ and $x_{s+1} = +\infty$. Show that for all $i = 0, 1, \dots, s$, the following properties hold:
    \begin{enumerate}
        \item For all $x\in (x_{i},x_{i+1})$, $\lbrace y \in \mathbb{R} : P(x,y) = 0 \rbrace$ is finite.
        \item For all $(x,y)\in (x_{i},x_{i+1})\times \mathbb{R}$ such that $P(x,y) = 0$, there exists an open neighborhood \(I_{x}\) of \(x\) and a unique function \(g \in \mathcal{C}^{1}(I_{x})\) such that \(g(x) = y\) and \(\forall x^\prime \in I_{x}\), \(P(x^\prime,g(x^\prime))=0\).
    \end{enumerate}

    Let $x\in(x_{i},x_{i+1})$, $t=\#\lbrace y \in \mathbb{R} : P(x,y) = 0 \rbrace$, and let $y_{1},y_{2},\dots,y_{t}$ be the roots of $P(x,y)$ in ascending order. Show that the following properties hold:
    \begin{enumerate}
        \setcounter{enumi}{2}
        \item For all $j=1,2,\dots,t$, let $g_j$ be the local function given by the previous lemma at $(x,y_{j})$. Show that $g_j$ extends uniquely to a continuous function \(h_{j}: (x_{i},x_{i+1}) \to \mathbb{R}\) such that \(h_{j}(x) = y_{j}\) and for all \(x^\prime \in (x_{i},x_{i+1})\), \(P(x^\prime,h_{j}(x^\prime))=0\).
        \item In addition, $h_{1}(x^\prime) < h_{2}(x^\prime) < \dots < h_{t}(x^\prime)$ for all $x^\prime\in (x_{i},x_{i+1})$.
        \item For all $x^\prime\in (x_{i},x_{i+1})$, $\#\lbrace y \in \mathbb{R} : P(x^\prime,y) = 0 \rbrace \geq t$.
    \end{enumerate}

    Finally, show that the number of roots of $P(x,y)$ in $\mathbb{R}$ is constant for all $x\in (x_{i},x_{i+1})$.
\end{exercise}

\begin{proof}
    \begin{enumerate}
        \item  For property 1, We know that \(\forall x \in (x_{i},x_{i+1})\), \(lc_{y}(P)(x) \neq 0\), so the polynomial in y \(P(x,y)\) isn't the zero polynomial. It's impossible that \(P(x,y)\) has an infinite number of roots. In conclusion, $\lbrace y \in \mathbb{R} : P(x,y) = 0 \rbrace$ is finite.
        \item  For property 2, \(\forall (x, y) \in (x_{i},x_{i+1}) \times \mathbb{R}\), such that \(P(x,y)=0\),we have \\\(Disc_{y}(P)(x) \neq 0\), so \(P(x,Y)\) don't have multiple roots, and \(P(x,y)=0\),so \(\frac{\partial P}{\partial y}(x,y) \neq 0\) P is polynomial function, so it's continuously differentiable. According to the Implicit Function Theorem, there exists an open neighborhood \(I_{x}\) of x, and a unique function \(g \in \mathcal{C}^{1}(I_{x})\), such that \(g(x) = y\) and \(\forall x' \in I_{x}\), \(P(x', g(x')) =0\).
        \item  For property 3, Fix \(j \in \{1,2,\dots,t\}.\) By property 2, at the point \((x,y_j)\) there is an open interval \(I_0\subset (x_i,x_{i+1})\) containing \(x\), and there is a function \(g_j:I_0 \to \mathbb{R}\) such that \(g_j(x)=y_j\) and \(P(u,g_j(u))=0\) for all \(u\in I_0\).
               We want to extend \(g_j\) as far as possible. Let \(h:I\to \mathbb{R}\) be a maximal extension of \(g_j\), where \(I\subset (x_i,x_{i+1})\) is an open Interval containing \(x\). This means that \(h(u)=g_j(u)\) for all \(u\in I\cap I_0\), and \(P(u,h(u))=0\) for all \(u\in I\). Also, \(I\) is maximal with this property. We will prove that \(I=(x_i,x_{i+1}).\) Suppose not. We first consider the right endpoint of \(I\). Write \(I=(\alpha,\beta).\) Assume, for contradiction, that \(\beta<x_{i+1}.\) Then \(\beta\in (x_i,x_{i+1}).\) Since there is no projection point in \((x_i,x_{i+1})\), we have \(\operatorname{lc}_y(P)(\beta)\neq 0\) and \(\operatorname{Disc}_y(P)(\beta)\neq 0.\) Write \(P\) as a polynomial in \(y\): \(P(u,y)=a_d(u)y^d+a_{d-1}(u)y^{d-1}+\cdots+a_0(u),\) where \(a_d(u)=\operatorname{lc}_y(P)(u).\) Since \(a_d(\beta)\neq 0,\)
               there is a small open interval \(U\) containing \(\beta\), and there is a
               constant \(c>0\), such that \(|a_d(u)|\geq c\) for all \(u\in U\). Also, the functions \(a_0(u),a_1(u),\dots,a_{d-1}(u)\) are continuous, so they are bounded on a smaller interval around \(\beta\). Therefore, the roots of \(P(u,Y)\) are uniformly bounded for \(u\) close to \(\beta\). Since \(P(u,h(u))=0\) for all \(u\in I\), this shows that \(h(u)\) is bounded when \(u\) is close to \(\beta\) from the left. Now choose a sequence \(u_n\in I\) such that \(u_n\to \beta\) and \(u_n<\beta.\) Since \(h(u_n)\) is bounded, we may take a subsequence such that \(h(u_n)\to y_\ast\)
               for some \(y_\ast\in\mathbb{R}.\) For simplicity, we still write this subsequence as \(u_n\). For every \(n\), we have \(P(u_n,h(u_n))=0.\) Taking the limit and using the continuity of \(P\), we get \(P(\beta,y_\ast)=0.\) Since \(\operatorname{Disc}_y(P)(\beta)\neq 0,\) the polynomial \(P(\beta,Y)\) has no multiple root. Hence \(y_\ast\) is a simple root of \(P(\beta,Y)\). Therefore \(\frac{\partial P}{\partial y}(\beta,y_\ast)\neq 0.\) By the implicit function theorem, there is an open interval \(K\) containing \(\beta\), and there is a function \(\varphi:K\to \mathbb{R}\) such that \(\varphi(\beta)=y_\ast\) and \(P(u,\varphi(u))=0\) for all \(u\in K\). We claim that \(h\) and \(\varphi\) are equal on a small interval to the left of \(\beta\). Indeed, since \((u_n,h(u_n))\to (\beta,y_\ast),\) the points \((u_n,h(u_n))\) are eventually inside the neighborhood where the implicit function theorem gives a unique local branch. Hence, for all sufficiently large \(n\), \(h(u_n)=\varphi(u_n).\) By the local uniqueness in the implicit function theorem, this equality extends to a small interval around such a point. Thus there exists \(\varepsilon>0\)such that \(h(u)=\varphi(u)\) for all \(u\in(\beta-\varepsilon,\beta).\) Now by the pasting lemma we can define a new function \(\widetilde h:I\cup K\to \mathbb{R}\) by \(\widetilde h(u)=\begin{cases}h(u), & u\in I,\\\varphi(u), & u\in K.\end{cases}\)
               This definition is well-defined because \(h\) and \(\varphi\) are equal on the overlap near \(\beta\). Also, \(P(u,\widetilde h(u))=0\) for all \(u\in I\cup K.\) But \(K\) contains points larger than \(\beta\), so \(I\cup K\) is strictly larger than \(I\). This contradicts the maximality of \(I\). Therefore the right endpoint of \(I\) cannot lie inside \((x_i,x_{i+1}).\) So we must have \(\beta=x_{i+1}\) if \(x_{i+1}\) is finite, and \(\beta=+\infty\) if \(x_{i+1}=+\infty\). The same argument at the left endpoint shows that the left endpoint of \(I\) cannot lie inside \((x_i,x_{i+1}).\) Hence \(I=(x_i,x_{i+1}).\) Therefore \(g_j\) extends to a continuous function \(h_j:(x_i,x_{i+1})\to \mathbb{R}\) such that \(h_j(x)=y_j\) and \(P(u,h_j(u))=0\) for all \(u\in (x_i,x_{i+1}).\) It remains to prove uniqueness. Suppose that \(\widetilde h_j:(x_i,x_{i+1})\to \mathbb{R}\) is another continuous extension of \(g_j\) such that \(P(u,\widetilde h_j(u))=0\) for all \(u\in (x_i,x_{i+1})\). Define \(A=\{u\in (x_i,x_{i+1}):h_j(u)=\widetilde h_j(u)\}.\)
               Since both functions extend \(g_j\), the set \(A\) is nonempty. Also, since \(h_j\) and \(\widetilde h_j\) are continuous, \(A\) is closed. Now let \(u_0\in A\), and set \(v_0=h_j(u_0)=\widetilde h_j(u_0). \) Since \(u_0\in (x_i,x_{i+1}),\) we have \(\operatorname{Disc}_y(P)(u_0)\neq 0.\)Thus \(\frac{\partial P}{\partial y}(u_0,v_0)\neq 0. \) By the local uniqueness in the implicit function theorem, \(h_j\) and \(\widetilde h_j\) are equal in a small neighborhood of \(u_0\). Hence \(A\) is open.So \(A\) is nonempty, open and closed in the interval \((x_i,x_{i+1}).\) Since this interval is connected, we get \(A=(x_i,x_{i+1}).\) Therefore \(h_j=\widetilde h_j.\)This proves uniqueness.
        \item  For property 4, suppose that there exists an index \(x' \in (x_{i}, x_{i+1})\), such that there exists \(h_{j1}, h_{j2}\) and \(h_{j1}(x') >= h_{j2}(x')\), and \(h_{j1}(x')=y_{1}<=h_{j2}(x')=y_{2}\), so \(h_{j1}(x')=h_{j2}(x')\), so \(x'\) is a root multiple, so \(Disc_{y}(x') \neq 0\). But \(x' \in (x_{i},x_{i+1})\), contradiction, so \(\forall x'\in (x_{i},x_{i+1})\), \(h_{1}(x') < h_{2}(x') < \dots < h_{t}(x')\)
        \item  For property 5, Let \(x'\in (x_i,x_{i+1})\)be fixed. By property 3, for each \(j=1,2,\dots,t,\) there exists a continuous function \(h_j:(x_i,x_{i+1})\to \mathbb{R}\) such that \(P(x',h_j(x'))=0.\) Therefore \(h_1(x'),h_2(x'),\dots,h_t(x')\) are real roots of the polynomial \(P(x',Y).\) By property 4, we have \(h_1(x')<h_2(x')<\cdots<h_t(x').\) Thus these \(t\) roots are pairwise different. Hence the polynomial \(P(x',Y)\)has at least \(t\) distinct real roots. Therefore \(\#\{y\in\mathbb{R}:P(x',y)=0\}\geq t.\) Since \(x'\) was arbitrary, the result holds for all \(x'\in (x_i,x_{i+1}).\)
        \item  Take two arbitrary points \(a,b\in (x_i,x_{i+1}).\) Let \(t_a=\#\{y\in\mathbb{R}:P(a,y)=0\}\) and \(t_b=\#\{y\in\mathbb{R}:P(b,y)=0\}.\) First, apply property 5 with the initial point \(a\). Then for every point in \((x_i,x_{i+1}),\)the number of real roots is at least \(t_a\). In particular, at the point \(b\), we get \(t_b\geq t_a.\) Next, apply property 5 with the initial point \(b\). Then for every point in \((x_i,x_{i+1}),\) the number of real roots is at least \(t_b\). In particular, at the point \(a\), we get \(t_a\geq t_b.\) Hence \(t_a=t_b.\)Since \(a\) and \(b\) were arbitrary, the number \(\#\{y\in\mathbb{R}:P(x,y)=0\}\) is the same for all \(x\in (x_i,x_{i+1}).\) Therefore the number of real roots of \(P(x,Y)\) is constant on \((x_i,x_{i+1}).\)
    \end{enumerate}
\end{proof}
